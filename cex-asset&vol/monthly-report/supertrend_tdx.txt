TR0:=MAX(MAX(H-L,ABS(H-REF(C,1))),ABS(L-REF(C,1)));
TR1:=IF(BARSCOUNT(C)=1,H-L,TR0);
AT1:=IF(BARSCOUNT(C)<N,DRAWNULL,SMA(TR1,N,1));
MIDP:=(H+L)/2;
BUB:=MIDP+M*AT1;
BLB:=MIDP-M*AT1;
FUP:=IF(ISNULL(AT1),DRAWNULL,IF(ISNULL(REF(FUP,1)),BUB,IF(REF(C,1)>REF(FUP,1),BUB,MIN(BUB,REF(FUP,1)))));
FDN:=IF(ISNULL(AT1),DRAWNULL,IF(ISNULL(REF(FDN,1)),BLB,IF(REF(C,1)<REF(FDN,1),BLB,MAX(BLB,REF(FDN,1)))));
DIR:=IF(ISNULL(AT1) OR ISNULL(REF(FUP,1)),-1,IF(REF(DIR,1)>=0,IF(C>=FDN,1,-1),IF(C<=FUP,-1,1)));
UPST:IF(DIR>0,FDN,DRAWNULL),COLORGREEN,LINETHICK2;
DNST:IF(DIR<0,FUP,DRAWNULL),COLORRED,LINETHICK2;
